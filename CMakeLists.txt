cmake_minimum_required(VERSION 3.16)
project(pao_system LANGUAGES CXX VERSION 1.0.0)

set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
if(NOT CMAKE_BUILD_TYPE)
  set(CMAKE_BUILD_TYPE Release)
endif()
if(CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang")
  add_compile_options(-Wall -Wextra -Wpedantic)
endif()

find_package(Threads REQUIRED)

add_library(paocore STATIC cpp/src/engine.cpp cpp/src/cascade.cpp)
target_include_directories(paocore PUBLIC cpp/include)
target_link_libraries(paocore PUBLIC Threads::Threads)

add_executable(pao_engine cpp/src/main.cpp)
target_link_libraries(pao_engine PRIVATE paocore)

enable_testing()
add_executable(pao_tests cpp/tests/test_engine.cpp)
target_link_libraries(pao_tests PRIVATE paocore)
add_test(NAME cpp_core COMMAND pao_tests)

# Необов'язковий Python-модуль (pip install pybind11)
option(PAO_PYTHON "Build the _paocore Python module" ON)
if(PAO_PYTHON)
  find_package(Python3 COMPONENTS Interpreter Development.Module QUIET)
  if(Python3_FOUND)
    execute_process(COMMAND ${Python3_EXECUTABLE} -c "import pybind11; print(pybind11.get_cmake_dir())"
                    OUTPUT_VARIABLE pybind11_DIR OUTPUT_STRIP_TRAILING_WHITESPACE ERROR_QUIET)
  endif()
  find_package(pybind11 CONFIG QUIET)
  if(pybind11_FOUND)
    pybind11_add_module(_paocore cpp/src/bindings.cpp)
    target_link_libraries(_paocore PRIVATE paocore)
    set_target_properties(_paocore PROPERTIES LIBRARY_OUTPUT_DIRECTORY ${CMAKE_SOURCE_DIR}/python/pao)
    message(STATUS "PAO: Python-модуль _paocore буде зібрано")
  else()
    message(STATUS "PAO: pybind11 не знайдено, Python працюватиме через pao_engine")
  endif()
endif()
